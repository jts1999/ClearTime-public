let type="lager";
let drinks=[];

const brand=document.getElementById("brand");
const serving=document.getElementById("serving");
const volume=document.getElementById("volume");
const abv=document.getElementById("abv");

function loadType(newType){

type=newType;

document.querySelectorAll(".types button").forEach(button=>{
button.classList.toggle("active",button.dataset.type===type);
});

brand.innerHTML="";

drinkData[type].brands.forEach(([name,strength])=>{

const option=document.createElement("option");

option.value=name;
option.textContent=name+" — "+strength+"%";
option.dataset.abv=strength;

brand.appendChild(option);

});

serving.innerHTML="";

drinkData[type].servings.forEach(([label,ml])=>{

const option=document.createElement("option");

option.value=ml;
option.textContent=label;

serving.appendChild(option);

});

if(type==="lager" || type==="cider"){
serving.value="568";
volume.value=568;
}else{
volume.value=drinkData[type].servings[0][1];
}

abv.value=brand.options[0].dataset.abv;

updateUnits();

}

document.querySelectorAll(".types button").forEach(button=>{

button.addEventListener("click",()=>{

loadType(button.dataset.type);

});

});

brand.addEventListener("change",()=>{

abv.value=brand.options[brand.selectedIndex].dataset.abv;

updateUnits();

});

serving.addEventListener("change",()=>{

volume.value=serving.value;

updateUnits();

});

volume.addEventListener("input",updateUnits);

abv.addEventListener("input",updateUnits);

function updateUnits(){

const units=(Number(volume.value)*Number(abv.value))/1000;

document.getElementById("singleUnits").textContent=
isFinite(units)?units.toFixed(2):"0.00";

}

document.getElementById("addDrink").addEventListener("click",()=>{

const v=Number(volume.value);
const a=Number(abv.value);

if(!v || !a) return;

drinks.push({
name:brand.value,
volume:v,
abv:a,
units:v*a/1000
});

render();

});

function render(){

const total=drinks.reduce((sum,d)=>sum+d.units,0);

document.getElementById("drinkCount").textContent=drinks.length;

document.getElementById("totalUnits").textContent=total.toFixed(2);

if(total){

const minutes=Math.ceil(total*60);

const time=new Date(Date.now()+minutes*60000);

document.getElementById("alcoholFree").textContent=
time.toLocaleTimeString([],{
hour:"2-digit",
minute:"2-digit"
});

}else{

document.getElementById("alcoholFree").textContent="—";

}

const history=document.getElementById("history");

history.innerHTML="";

drinks.forEach((drink,index)=>{

const row=document.createElement("div");

row.className="drinkrow";

const info=document.createElement("div");

info.innerHTML=
"<strong>"+drink.name+"</strong>"+
"<div class='muted'>"+
drink.volume+" ml · "+
drink.abv+"% · "+
drink.units.toFixed(2)+" units"+
"</div>";

const remove=document.createElement("button");

remove.textContent="Remove";

remove.onclick=()=>{

drinks.splice(index,1);

render();

};

row.appendChild(info);

row.appendChild(remove);

history.appendChild(row);

});

}

document.getElementById("reset").onclick=()=>{

drinks=[];

render();

};

document.getElementById("ftBtn").onclick=()=>{

const cm=Number(document.getElementById("heightCm").value);

const inches=cm/2.54;

document.getElementById("heightFt").value=
Math.floor(inches/12);

document.getElementById("heightIn").value=
Math.round(inches%12);

document.getElementById("cmHeight").classList.add("hidden");

document.getElementById("ftHeight").classList.remove("hidden");

};

document.getElementById("cmBtn").onclick=()=>{

const ft=Number(document.getElementById("heightFt").value);

const inches=Number(document.getElementById("heightIn").value);

document.getElementById("heightCm").value=
Math.round((ft*12+inches)*2.54);

document.getElementById("ftHeight").classList.add("hidden");

document.getElementById("cmHeight").classList.remove("hidden");

};

loadType("lager");

render();
